Standar keamanan yang benar dan wajib untuk sistem absensi sungguhan. Keputusan di sisi klien memang tidak boleh dipercaya. 
 
🔐 Sistem Absen Digital — Versi Produksi Aman
 
Prinsip Utama
 
Frontend hanya mengumpulkan data. SERVER yang memutuskan.
Tidak ada  boleh = true  di JavaScript. Semua validasi, perhitungan jarak, waktu, dan status dilakukan di sisi backend.
 
 
 
📦 Struktur Sistem
 
1. Frontend → ambil GPS, ambil foto, kirim data mentah
​
2. Firebase Auth → login pengguna
​
3. Firebase Storage → simpan foto (bukan base64 di database)
​
4. Firestore/RTDB → simpan data absen
​
5. Cloud Function → validasi, hitung jarak, cek duplikat, putuskan diterima/ditolak

1️⃣ Frontend —  absen.html
2️⃣ Cloud Function —  functions/index.js  (Server-Side Validation)
 
Ini otak keamanan — semua keputusan di sini, tidak bisa diubah pengguna:
const functions = require('firebase-functions/v1');
const admin = require('firebase-admin');
admin.initializeApp();
const db = admin.firestore();

// ========= KONFIGURASI KANTOR =========
const KANTOR = {
    lat: -7.2935,    // Ganti koordinat kantor asli
    lng: 112.7935,
    radiusMeter: 100, // Jarak maksimal diizinkan
    akurasiMaksimal: 50 // Akurasi GPS buruk ditolak
};

// Hitung jarak — dilakukan di server
function hitungJarak(lat1, lon1, lat2, lon2) {
    const R = 6371000;
    const dLat = (lat2 - lat1) * Math.PI / 180;
    const dLon = (lon2 - lon1) * Math.PI / 180;
    const a = Math.sin(dLat/2)**2 + 
              Math.cos(lat1 * Math.PI / 180) * 
              Math.cos(lat2 * Math.PI / 180) * 
              Math.sin(dLon/2)**2;
    return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

// Endpoint dipanggil dari frontend
exports.validasiAbsen = functions.https.onRequest(async (req, res) => {
    res.set('Access-Control-Allow-Origin', '*');
    if (req.method === 'OPTIONS') {
        res.set('Access-Control-Allow-Methods', 'POST');
        res.set('Access-Control-Allow-Headers', 'Content-Type');
        return res.status(204).send('');
    }

    const { uid, email, nama, lat, lng, akurasi, urlFoto } = req.body;

    // 1. Validasi data lengkap
    if (!uid || !lat || !lng || !urlFoto) {
        return res.json({ diterima: false, pesan: 'Data tidak lengkap', alasan: 'Ada data yang hilang' });
    }

    // 2. Cek akurasi GPS
    if (akurasi > KANTOR.akurasiMaksimal) {
        return res.json({
            diterima: false,
            pesan: 'Sinyal GPS kurang akurat',
            alasan: `Akurasi: ${Math.round(akurasi)} m, maksimal: ${KANTOR.akurasiMaksimal} m. Pindah ke tempat terbuka.`
        });
    }

    // 3. Hitung jarak — DI SERVER, tidak bisa diubah pengguna
    const jarak = hitungJarak(lat, lng, KANTOR.lat, KANTOR.lng);
    if (jarak > KANTOR.radiusMeter) {
        return res.json({
            diterima: false,
            pesan: 'Di luar area absen',
            alasan: `Jarak: ${Math.round(jarak)} m, batas: ${KANTOR.radiusMeter} m`
        });
    }

    // 4. Cek sudah absen hari ini — cegah ganda
    const hariIni = new Date().toISOString().slice(0, 10);
    const cekSudahAbsen = await db
        .collection('absen')
        .where('uid', '==', uid)
        .where('tanggal', '==', hariIni)
        .limit(1)
        .get();

    if (!cekSudahAbsen.empty) {
        return res.json({
            diterima: false,
            pesan: 'Sudah absen hari ini',
            alasan: 'Tidak dapat melakukan absen ulang'
        });
    }

    // 5. Semua syarat terpenuhi → SIMPAN DENGAN WAKTU SERVER
    const dataAbsen = {
        uid: uid,
        nama: nama,
        email: email,
        tanggal: hariIni,
        waktuServer: admin.firestore.FieldValue.serverTimestamp(), // ⭐ Waktu dari server!
        lokasi: {
            lat: lat,
            lng: lng,
            akurasi: akurasi,
            jarakKeKantor: Math.round(jarak)
        },
        urlFoto: urlFoto,
        ipPengguna: req.ip // Catat IP untuk audit
    };

    await db.collection('absen').add(dataAbsen);

    return res.json({
        diterima: true,
        pesan: `Selamat! Absen tercatat. Jarak: ${Math.round(jarak)} m`,
        waktuServer: new Date().toLocaleString('id-ID', { timeZone: 'Asia/Jakarta' })
    });
});


3️⃣ Firebase Security Rules —  firestore.rules 
 
Mencegah pengguna mengubah data orang lain:
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Baca absen sendiri saja
    match /absen/{id} {
      allow read: if request.auth != null && request.auth.uid == resource.data.uid;
      allow create: if false; // HANYA Cloud Function yang boleh simpan!
      allow update, delete: if false;
    }
  }
}

Storage Rules ( storage.rules )

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /absen/{uid}/{allPaths=**} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if request.auth != null && request.auth.uid == uid;
    }
  }
}
